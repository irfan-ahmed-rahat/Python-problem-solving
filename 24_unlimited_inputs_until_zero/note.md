print("Enter numbers (Enter 0 to stop):")

#ধাপ ১: একটি অসীম লুপ (Infinite Loop) শুরু করা
while True:
    
#ধাপ ২: লুপের ভেতরে ইউজারের কাছ থেকে আনলিমিটেড বার ইনপুট নেওয়া
    num = int(input("Enter a number: "))

    if num == 0:
        print("You entered 0. Exiting the loop!")
        # ধাপ ৪: 0 পেলে লুপ ভেঙে বেরিয়ে আসার কমান্ড দেওয়া
        break 
        
#0 না হলে সংখ্যাটি প্রিন্ট করে আবার নতুন ইনপুটের জন্য লুপের শুরুতে চলে যাবে
    print(f"You entered: {num}")

    #শেষের print লাইনটি if এর সোজাসুজি থাকার মানে হলো— সে if এর ভেতরে নেই।
